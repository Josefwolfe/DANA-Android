cmake_minimum_required(VERSION 3.22.1)
project(dana_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(WHISPER_OPENMP OFF CACHE BOOL "" FORCE)
if(NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/whisper.cpp/CMakeLists.txt")
    message(FATAL_ERROR "Pinned whisper.cpp source missing: run scripts/fetch_whisper_cpp.sh first")
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/whisper.cpp" "${CMAKE_CURRENT_BINARY_DIR}/whisper_upstream" EXCLUDE_FROM_ALL)
add_library(dana_whisper SHARED whisper_jni.cpp)
target_include_directories(dana_whisper PRIVATE "${CMAKE_CURRENT_LIST_DIR}/whisper.cpp/include")
target_link_libraries(dana_whisper PRIVATE whisper log)
